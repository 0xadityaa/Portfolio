   137	export async function getGitHubBuilderProfile(username: string = "0xadityaa"): Promise<GitHubBuilderProfile> {
   138	  const token = process.env.GITHUB_TOKEN;
   139	
   140	  if (!token || token.includes("dummy")) {
   141	    console.warn(`
   142	⚠️ [WARN] Missing or dummy GITHUB_TOKEN environment variable!
   143	Returning mock GitHub data.
   144	`);
   145	    return getMockProfile();
   146	  }
   147	
   148	  try {
   149	    const res = await fetch(GITHUB_GRAPHQL_API, {
   150	      method: "POST",
   151	      headers: {
   152	        Authorization: `Bearer ${token}`,
   153	        "Content-Type": "application/json",
   154	        "User-Agent": "antigravity-builder-portfolio",
   155	      },
   156	      body: JSON.stringify({
   157	        query,
   158	        variables: { username },
   159	      }),
   160	      next: {
   161	        revalidate: 3600, // Cache for 1 hour
   162	        tags: ["github-profile"],
   163	      },
   164	    });
   165	
   166	  if (!res.ok) {
   167	      const errorText = await res.text();
   168	      throw new Error(`GitHub GraphQL API responded with status ${res.status}: ${errorText}`);
   169	    }
   170	
   171	    const { data, errors } = await res.json();
   172	
   173	    if (errors && errors.length > 0) {
   174	      throw new Error(`GitHub GraphQL errors: ${errors.map((e: any) => e.message).join(", ")}`);
   175	    }
   176	
   177	    const user = data?.user;
   178	    if (!user) {
   179	      throw new Error(`GitHub user "${username}" not found.`);
   180	    }
   181	
   182	    const mapRepoNode = (node: any): GitHubPinnedRepo => ({
   183	      name: node.name,
   184	      description: node.description || "",
   185	      url: node.url,
   186	      homepageUrl: node.homepageUrl || null,
   187	      stargazerCount: node.stargazerCount || 0,
   188	      forkCount: node.forkCount || 0,
   189	      issuesCount: node.issues?.totalCount || 0,
   190	      commitsCount: node.defaultBranchRef?.target?.history?.totalCount || 0,
   191	      openGraphImageUrl: node.openGraphImageUrl || "",
   192	      pushedAt: node.pushedAt || new Date().toISOString(),
   193	      primaryLanguage: node.primaryLanguage
   194	        ? {
   195	            name: node.primaryLanguage.name,
   196	            color: node.primaryLanguage.color,
   197	          }
   198	        : null,
   199	      languages: (node.languages?.nodes || []).map((l: any) => l.name),
   200	      topics: (node.repositoryTopics?.nodes || []).map((t: any) => t.topic.name),
   201	    });
   202	
   203	    const pinnedRepos: GitHubPinnedRepo[] = (user.pinnedItems?.nodes || []).map(mapRepoNode);
   204	    const repos: GitHubPinnedRepo[] = (user.repositories?.nodes || []).map(mapRepoNode);
   205	
   206	    // Sum all repository commits as an all-time contribution approximation
   207	    const allTimeCommits = repos.reduce((acc, repo) => acc + (repo.commitsCount || 0), 0);
   208	    const defaultContributions = user.contributionsCollection?.contributionCalendar?.totalContributions || 0;
   209	
   210	    return {
   211	      name: user.name || "Aditya Negandhi",
   212	      bio: user.bio || "Full Stack Engineer & aspiring Solutions Architect.",
   213	      company: user.company || null,
   214	      location: user.location || null,
   215	      avatarUrl: user.avatarUrl || "/images/profile/avatar.png",
   216	      followersCount: user.followers?.totalCount || 0,
   217	      publicReposCount: user.repositories?.totalCount || 0,
   218	      contributionsCount: allTimeCommits > 0 ? allTimeCommits : defaultContributions,
   219	      pinnedRepos,
   220	      repos,
   221	    };
   222	  } catch (error) {
   223	    console.error("❌ Failed to fetch GitHub profile stats:", error);
   224	    console.warn("⚠️ Returning mock GitHub data due to fetch failure.");
   225	    return getMockProfile();
   226	  }
   227	}
